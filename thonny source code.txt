


# Note - DHT22 data pin is GPIO 4


import network
import time
import dht
from machine import Pin
from umqtt.simple import MQTTClient


WIFI_SSID = "Bellanie Bay"
WIFI_PASSWORD = "savinggrace"

MQTT_BROKER = "192.168.132.68"

MQTT_PORT = 1883


MQTT_USER = None
MQTT_PASSWORD = None

MQTT_CLIENT_ID = "ESP32_DHT22"

MQTT_PUB_TEMP = b"esp32/dht/temperature"
MQTT_PUB_HUM = b"esp32/dht/humidity"


DHT_PIN = 4

sensor = dht.DHT22(Pin(DHT_PIN))


def connect_wifi():

    wlan = network.WLAN(network.STA_IF)
    wlan.active(True)

    if not wlan.isconnected():

        print("Connecting to Wi-Fi...")

        wlan.connect(WIFI_SSID, WIFI_PASSWORD)

        
        timeout = 20

        while not wlan.isconnected() and timeout > 0:
            print(".", end="")
            time.sleep(1)
            timeout -= 1

        print()

    if wlan.isconnected():

        print("Wi-Fi connected")
        print("IP address:", wlan.ifconfig()[0])

        return wlan

    else:

        print("Wi-Fi connection failed")
        return None




def connect_mqtt():

    print("Connecting to MQTT...")

    try:

        client = MQTTClient(
            MQTT_CLIENT_ID,
            MQTT_BROKER,
            port=MQTT_PORT,
            user=MQTT_USER,
            password=MQTT_PASSWORD
        )

        client.connect()

        print("Connected to MQTT")

        return client

    except Exception as e:

        print("MQTT connection failed:", e)

        return None




print()
print("==============================")
print("ESP32 DHT22 MQTT")
print("==============================")


wlan = connect_wifi()

if wlan is None:
    print("Cannot connect to Wi-Fi.")
else:

    
    mqtt_client = connect_mqtt()

    if mqtt_client is None:
        print("Cannot connect to MQTT.")

    else:

        last_reading = time.ticks_ms()

        while True:

            try:

                # Check Wi-Fi connection
                if not wlan.isconnected():

                    print("Wi-Fi connection lost!")
                    print("Reconnecting...")

                    wlan = connect_wifi()

                    if wlan is None:
                        time.sleep(5)
                        continue

                    mqtt_client = connect_mqtt()

                    if mqtt_client is None:
                        time.sleep(5)
                        continue

                # Publish every 5 seconds
                if time.ticks_diff(
                    time.ticks_ms(),
                    last_reading
                ) >= 5000:

                    last_reading = time.ticks_ms()

                    # Read DHT22
                    sensor.measure()

                    temperature = sensor.temperature()
                    humidity = sensor.humidity()

                    print()
                    print("------------------------------")
                    print("Temperature:", temperature, "C")
                    print("Humidity:", humidity, "%")

                    
                    temp_message = "{:.2f}".format(temperature)
                    hum_message = "{:.2f}".format(humidity)

                    # Publish temperature
                    mqtt_client.publish(
                        MQTT_PUB_TEMP,
                        temp_message.encode(),
                        retain=True,
                        qos=1
                    )

                    print(
                        "Published temperature:",
                        temp_message
                    )

                    # Publish humidity
                    mqtt_client.publish(
                        MQTT_PUB_HUM,
                        hum_message.encode(),
                        retain=True,
                        qos=1
                    )

                    print(
                        "Published humidity:",
                        hum_message
                    )

                # Small delay
                time.sleep(1)

            except OSError as e:

                print()
                print("Sensor/MQTT error:", e)
                print("Trying again...")

                time.sleep(5)

                
                mqtt_client = connect_mqtt()

            except Exception as e:

                print()
                print("Error:", e)

                time.sleep(5)
